import { Router, Request, Response } from 'express';
import { databaseService } from '../services/database.service';
import { logger } from '../utils/logger';

const router = Router();

interface HealthStatus {
  status: 'healthy' | 'degraded' | 'unhealthy';
  timestamp: string;
  checks: Record<string, {
    status: 'pass' | 'fail';
    message: string;
    responseTime: number;
  }>;
  uptime: number;
}

const startTime = Date.now();

/**
 * GET /health
 * Quick health check (< 1s)
 */
router.get('/health', async (req: Request, res: Response) => {
  try {
    const checks: Record<string, any> = {};

    // 1. Database connectivity
    const dbStart = Date.now();
    try {
      await databaseService.query('SELECT 1');
      checks.database = {
        status: 'pass',
        message: 'Connected',
        responseTime: Date.now() - dbStart,
      };
    } catch (error) {
      checks.database = {
        status: 'fail',
        message: (error as Error).message,
        responseTime: Date.now() - dbStart,
      };
    }

    // 2. Table checks
    const tableStart = Date.now();
    try {
      const tables = ['users', 'billing_records', 'usage_audit'];
      const result = await databaseService.query(
        `SELECT COUNT(*) as count FROM information_schema.tables 
         WHERE table_name IN (${tables.map((_, i) => `$${i + 1}`).join(',')})`,
        tables
      );

      checks.tables = {
        status: parseInt(result.rows[0].count) === tables.length ? 'pass' : 'fail',
        message: `${result.rows[0].count}/${tables.length} tables found`,
        responseTime: Date.now() - tableStart,
      };
    } catch (error) {
      checks.tables = {
        status: 'fail',
        message: (error as Error).message,
        responseTime: Date.now() - tableStart,
      };
    }

    // Overall status
    const allPassed = Object.values(checks).every((c: any) => c.status === 'pass');

    const response: HealthStatus = {
      status: allPassed ? 'healthy' : 'degraded',
      timestamp: new Date().toISOString(),
      checks,
      uptime: Math.floor((Date.now() - startTime) / 1000), // seconds
    };

    res.status(allPassed ? 200 : 503).json(response);
  } catch (error) {
    logger.error('Health check failed', error);
    res.status(500).json({
      status: 'unhealthy',
      error: (error as Error).message,
    });
  }
});

/**
 * GET /health/detailed
 * Detailed health check (1-5s)
 */
router.get('/health/detailed', async (req: Request, res: Response) => {
  try {
    const checks: Record<string, any> = {};

    // Database
    const dbStart = Date.now();
    try {
      const result = await databaseService.query('SELECT COUNT(*) as count FROM users');
      checks.database = {
        status: 'pass',
        message: 'Connected',
        users: parseInt(result.rows[0].count),
        responseTime: Date.now() - dbStart,
      };
    } catch (error) {
      checks.database = {
        status: 'fail',
        message: (error as Error).message,
        responseTime: Date.now() - dbStart,
      };
    }

    // Billing records
    const billingStart = Date.now();
    try {
      const result = await databaseService.query(
        'SELECT COUNT(*) as count FROM billing_records WHERE status = $1',
        ['pending']
      );
      checks.billing = {
        status: 'pass',
        message: 'Table accessible',
        pendingRecords: parseInt(result.rows[0].count),
        responseTime: Date.now() - billingStart,
      };
    } catch (error) {
      checks.billing = {
        status: 'fail',
        message: (error as Error).message,
        responseTime: Date.now() - billingStart,
      };
    }

    // Usage audit
    const auditStart = Date.now();
    try {
      const result = await databaseService.query(
        `SELECT COUNT(*) as count FROM usage_audit 
         WHERE recorded_at > NOW() - INTERVAL '24 hours'`
      );
      checks.audit = {
        status: 'pass',
        message: '24h coverage',
        last24Hours: parseInt(result.rows[0].count),
        responseTime: Date.now() - auditStart,
      };
    } catch (error) {
      checks.audit = {
        status: 'fail',
        message: (error as Error).message,
        responseTime: Date.now() - auditStart,
      };
    }

    const allPassed = Object.values(checks).every((c: any) => c.status === 'pass');

    res.status(allPassed ? 200 : 503).json({
      status: allPassed ? 'healthy' : 'degraded',
      timestamp: new Date().toISOString(),
      checks,
      uptime: Math.floor((Date.now() - startTime) / 1000),
    });
  } catch (error) {
    logger.error('Detailed health check failed', error);
    res.status(500).json({
      status: 'unhealthy',
      error: (error as Error).message,
    });
  }
});

export default router;
