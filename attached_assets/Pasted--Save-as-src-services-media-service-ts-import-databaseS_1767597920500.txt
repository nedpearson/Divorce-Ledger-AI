import { databaseService } from './database.service';
import { logger } from '../utils/logger';

export interface MediaFile {
  id: string;
  violation_id: number;
  case_id: number;
  file_type: 'audio' | 'video' | 'image';
  file_name: string;
  file_size_bytes: number;
  mime_type: string;
  storage_url: string;
  duration_seconds?: number;
  transcript?: string;
  transcript_confidence?: number;
  processing_status: 'pending' | 'processing' | 'completed' | 'failed';
  created_at: Date;
}

export interface ViolationMediaData {
  audio_transcript?: string;
  media_files: MediaFile[];
  ai_classification?: string;
  severity_score?: number;
  processing_metadata: {
    total_duration?: number;
    media_count: number;
    transcription_confidence?: number;
    key_phrases?: string[];
  };
}

export class MediaService {
  /**
   * Upload voice/media file to violation
   */
  async uploadMedia(
    violationId: number,
    caseId: number,
    fileName: string,
    fileType: 'audio' | 'video' | 'image',
    fileSizeBytes: number,
    mimeType: string,
    storageUrl: string,
    durationSeconds?: number
  ): Promise<MediaFile> {
    try {
      // Validate file size based on user tier
      const violation = await databaseService.query(
        `SELECT v.*, c.lawyer_id FROM violations v
         JOIN cases c ON v.case_id = c.id
         WHERE v.id = $1`,
        [violationId]
      );

      if (!violation.rows[0]) {
        throw new Error(`Violation ${violationId} not found`);
      }

      const user = await databaseService.query(
        'SELECT tier FROM users WHERE id = $1',
        [violation.rows[0].lawyer_id]
      );

      const maxFileSize = this.getMaxFileSizeForTier(user.rows[0]?.tier || 'free');

      if (fileSizeBytes > maxFileSize) {
        throw new Error(
          `File exceeds ${maxFileSize / 1024 / 1024}MB limit for ${user.rows[0]?.tier || 'free'} tier`
        );
      }

      // Insert into evidence table
      const result = await databaseService.query(
        `INSERT INTO evidence (
          violation_id, case_id, file_type, file_name, file_size_bytes, 
          mime_type, storage_url, evidence_source, evidence_metadata, created_at, updated_at
        ) VALUES ($1, $2, $3, $4, $5, $6, $7, 'media_upload', $8, NOW(), NOW())
         RETURNING *`,
        [
          violationId,
          caseId,
          fileType,
          fileName,
          fileSizeBytes,
          mimeType,
          storageUrl,
          JSON.stringify({
            file_type: fileType,
            duration_seconds: durationSeconds,
            upload_timestamp: new Date().toISOString(),
            processing_status: 'pending',
          }),
        ]
      );

      const mediaFile = result.rows[0];

      // Queue transcription for audio/video
      if (fileType === 'audio' || fileType === 'video') {
        await this.queueForTranscription(violationId, mediaFile.id, fileType);
      }

      logger.info(\`✅ Media uploaded: \${fileName} (\${fileType})\`);
      return mediaFile;
    } catch (error) {
      logger.error('Media upload failed', error);
      throw error;
    }
  }

  private getMaxFileSizeForTier(tier: string): number {
    const sizes: Record<string, number> = {
      free: 10 * 1024 * 1024,
      pro: 100 * 1024 * 1024,
      enterprise: 500 * 1024 * 1024,
    };
    return sizes[tier] || sizes.free;
  }

  private async queueForTranscription(
    violationId: number,
    mediaFileId: string,
    mediaType: string
  ): Promise<void> {
    try {
      await databaseService.query(
        `INSERT INTO processing_queue (
          raw_document_id, job_type, status, priority, created_at, updated_at
        ) VALUES ($1, $2, 'pending', 3, NOW(), NOW())`,
        [mediaFileId, \`transcribe_\${mediaType}\`]
      );

      logger.info(\`✅ Queued transcription for media \${mediaFileId}\`);
    } catch (error) {
      logger.error('Failed to queue transcription', error);
    }
  }

  /**
   * Transcribe audio to text
   */
  async transcribeAudio(violationId: number, mediaFileId: string): Promise<string> {
    try {
      const mediaResult = await databaseService.query(
        `SELECT * FROM evidence WHERE id = $1 AND violation_id = $2`,
        [mediaFileId, violationId]
      );

      if (mediaResult.rows.length === 0) {
        throw new Error('Media file not found');
      }

      const media = mediaResult.rows[0];
      const transcript = await this.simulateTranscription(media.storage_url);

      // Update violation with transcript
      await databaseService.query(
        \`UPDATE violations 
         SET audio_transcript = $1, updated_at = NOW()
         WHERE id = $2\`,
        [transcript, violationId]
      );

      // Update evidence metadata with transcript
      await databaseService.query(
        \`UPDATE evidence 
         SET evidence_metadata = jsonb_set(
           COALESCE(evidence_metadata, '{}'::jsonb),
           '{transcript}',
           to_jsonb($1::text)
         ),
         updated_at = NOW()
         WHERE id = $2\`,
        [transcript, mediaFileId]
      );

      logger.info(\`✅ Audio transcribed for violation \${violationId}\`);
      return transcript;
    } catch (error) {
      logger.error('Transcription failed', error);
      throw error;
    }
  }

  private async simulateTranscription(audioUrl: string): Promise<string> {
    return \`[TRANSCRIBED AUDIO]
    
"You need to listen to me... I told you not to take the kids without my permission. 
This is the third time this week. I'm getting really frustrated with this situation. 
You can't just keep ignoring what we agreed to. This is affecting the children and it needs to stop."

[Timestamp: 00:15 - Tone: Elevated, Frustrated]
[Timestamp: 00:45 - Keywords: Permission, Agreement, Children, Frustrated]
[Source: \${audioUrl}]\`;
  }

  /**
   * AI classify violation from transcript
   */
  async classifyViolationFromMedia(violationId: number): Promise<any> {
    try {
      const violationResult = await databaseService.query(
        \`SELECT id, audio_transcript, description FROM violations WHERE id = $1\`,
        [violationId]
      );

      const violation = violationResult.rows[0];
      if (!violation) {
        throw new Error(\`Violation \${violationId} not found\`);
      }
      if (!violation.audio_transcript) {
        throw new Error('No transcript available for classification');
      }

      const classification = this.analyzeTranscriptForViolations(violation.audio_transcript);

      // Update violation with classification and severity
      await databaseService.query(
        \`UPDATE violations 
         SET ai_classification = $1,
             severity_score = $2,
             updated_at = NOW()
         WHERE id = $3\`,
        [classification.type, classification.severity, violationId]
      );

      logger.info(\`✅ Classified: \${classification.type} (Confidence: \${(classification.confidence * 100).toFixed(0)}%)\`);
      return classification;
    } catch (error) {
      logger.error('Classification failed', error);
      throw error;
    }
  }

  private analyzeTranscriptForViolations(transcript: string): any {
    const lowerTranscript = transcript.toLowerCase();

    const patterns = {
      harassment: {
        keywords: [
          'constant',
          'repeated',
          'won\\'t stop',
          'every day',
          'always',
          'can\\'t handle',
          'fed up',
        ],
        threshold: 0.6,
      },
      control: {
        keywords: ['need to listen', 'have to', 'must', 'i control', 'you can\\'t', 'permission'],
        threshold: 0.5,
      },
      threats: {
        keywords: [
          'going to',
          'you\\'ll regret',
          'i\\'ll make you',
          'you\\'ll see',
          'i\\'m warning you',
        ],
        threshold: 0.7,
      },
      verbal_abuse: {
        keywords: ['stupid', 'idiot', 'crazy', 'useless', 'pathetic', 'loser', 'worthless'],
        threshold: 0.5,
      },
    };

    const scores: Record<string, number> = {};
    let primaryType = 'unclassified';
    let maxConfidence = 0;

    for (const [type, pattern] of Object.entries(patterns)) {
      const matches = pattern.keywords.filter((keyword: string) =>
        lowerTranscript.includes(keyword)
      );
      const matchRate = matches.length / pattern.keywords.length;

      if (matchRate >= pattern.threshold) {
        scores[type] = matchRate;
        if (matchRate > maxConfidence) {
          maxConfidence = matchRate;
          primaryType = type;
        }
      }
    }

    return {
      type: primaryType,
      confidence: Math.round(maxConfidence * 100) / 100,
      detected_patterns: Object.keys(scores),
      severity: this.calculateSeverityFromClassification(primaryType, maxConfidence),
      flagged_keywords: Object.entries(scores).map(([type, score]) => ({
        type,
        score: Math.round(score * 100) / 100,
      })),
      recommendation: maxConfidence > 0.7 ? 'Flag for lawyer review' : 'Monitor for escalation',
    };
  }

  private calculateSeverityFromClassification(type: string, confidence: number): number {
    const baseSeverity: Record<string, number> = {
      harassment: 60,
      control: 55,
      threats: 85,
      verbal_abuse: 65,
      unclassified: 30,
    };

    const base = baseSeverity[type] || 30;
    return Math.round(base * confidence);
  }

  /**
   * Get all media and transcripts for violation
   */
  async getMediaForViolation(violationId: number): Promise<ViolationMediaData> {
    try {
      const result = await databaseService.query(
        \`SELECT 
          v.id,
          v.audio_transcript,
          v.ai_classification,
          v.severity_score,
          COALESCE(json_agg(json_build_object(
            'id', e.id,
            'file_type', e.file_type,
            'file_name', e.file_name,
            'file_size_bytes', e.file_size_bytes,
            'mime_type', e.mime_type,
            'storage_url', e.storage_url,
            'duration_seconds', (e.evidence_metadata->>'duration_seconds')::int,
            'processing_status', COALESCE(e.evidence_metadata->>'processing_status', 'completed')
          )) FILTER (WHERE e.id IS NOT NULL), '[]'::json) as media_files
         FROM violations v
         LEFT JOIN evidence e ON v.id = e.violation_id 
           AND e.evidence_source = 'media_upload'
         WHERE v.id = $1
         GROUP BY v.id, v.audio_transcript, v.ai_classification, v.severity_score\`,
        [violationId]
      );

      if (result.rows.length === 0) {
        throw new Error(\`Violation \${violationId} not found\`);
      }

      const row = result.rows[0];

      return {
        audio_transcript: row.audio_transcript,
        media_files: row.media_files && row.media_files.length > 0 ? row.media_files : [],
        ai_classification: row.ai_classification,
        severity_score: row.severity_score,
        processing_metadata: {
          media_count: row.media_files ? row.media_files.length : 0,
        },
      };
    } catch (error) {
      logger.error('Failed to get media for violation', error);
      throw error;
    }
  }

  /**
   * Get violations count for this month
   */
  async getViolationsThisMonth(caseId: number): Promise<number> {
    try {
      const result = await databaseService.query(
        \`SELECT COUNT(*) as count FROM violations 
         WHERE case_id = $1 
         AND DATE_TRUNC('month', created_at) = DATE_TRUNC('month', NOW())\`,
        [caseId]
      );

      return parseInt(result.rows[0].count);
    } catch (error) {
      logger.error('Failed to get monthly violation count', error);
      throw error;
    }
  }

  /**
   * Update user tier based on violation count
   */
  async updateUserTierMetrics(userId: number): Promise<void> {
    try {
      const caseIds = await databaseService.query(
        \`SELECT DISTINCT case_id FROM cases WHERE lawyer_id = $1\`,
        [userId]
      );

      let totalViolations = 0;

      for (const caseRow of caseIds.rows) {
        const count = await this.getViolationsThisMonth(caseRow.case_id);
        totalViolations += count;
      }

      // Determine tier
      let newTier = 'free';
      if (totalViolations > 50) {
        newTier = 'enterprise';
      } else if (totalViolations > 20) {
        newTier = 'pro';
      }

      await databaseService.query(
        \`UPDATE users 
         SET tier = $1, violations_count_this_month = $2, updated_at = NOW()
         WHERE id = $3\`,
        [newTier, totalViolations, userId]
      );

      logger.info(\`✅ User tier updated: \${newTier} (\${totalViolations} violations)\`);
    } catch (error) {
      logger.error('Failed to update user tier', error);
    }
  }
}

export const mediaService = new MediaService();
