  <script>
    // Initialize Socket.io connection
    const socket = io();
    let tierChart, billingChart;

    socket.on('connect', () => {
      console.log('✅ Connected to monitoring server');
      updateStatus('connected');
    });

    socket.on('metrics:initial', (metrics) => {
      console.log('📊 Received initial metrics:', metrics);
      updateDashboard(metrics);
    });

    socket.on('metrics:update', (metrics) => {
      console.log('🔄 Metrics updated:', metrics);
      updateDashboard(metrics);
    });

    socket.on('disconnect', () => {
      console.log('❌ Disconnected from server');
      updateStatus('disconnected');
    });

    function updateStatus(status) {
      const dot = document.getElementById('statusDot');
      const text = document.getElementById('statusText');

      if (status === 'connected') {
        dot.classList.remove('warning', 'error');
        text.textContent = 'Healthy';
        text.style.color = '#10b981';
      } else if (status === 'disconnected') {
        dot.classList.add('error');
        text.textContent = 'Disconnected';
        text.style.color = '#ef4444';
      } else {
        dot.classList.add('warning');
        text.textContent = 'Connecting...';
        text.style.color = '#f59e0b';
      }
    }

    function updateDashboard(metrics) {
      const formatDate = (dateStr) => {
        return new Date(dateStr).toLocaleString();
      };

      const formatUptime = (seconds) => {
        const hours = Math.floor(seconds / 3600);
        const minutes = Math.floor((seconds % 3600) / 60);
        if (hours > 0) return `${hours}h ${minutes}m`;
        return `${minutes}m`;
      };

      // Update KPI cards
      document.getElementById('totalUsers').textContent = metrics.users.total;
      document.getElementById('activeUsers7d').textContent = metrics.users.active_7d;

      document.getElementById('monthlyRevenue').textContent = `$${metrics.billing.charged_this_month_usd}`;
      document.getElementById('avgRevenue').textContent = metrics.users.total > 0 
        ? `$${Math.round(metrics.billing.charged_this_month_usd / metrics.users.total)}` 
        : '$0';

      document.getElementById('pendingBilling').textContent = metrics.billing.pending_count;
      document.getElementById('pendingAmount').textContent = metrics.billing.pending_amount_usd;

      document.getElementById('dbHealth').textContent = metrics.health.db_response_time_ms;
      document.getElementById('apiStatus').textContent = metrics.health.db_response_time_ms < 100 ? 'Optimal' : 'Good';

      document.getElementById('totalViolations').textContent = metrics.violations.total_this_month;
      document.getElementById('atRiskUsers').textContent = metrics.violations.at_risk_users;

      document.getElementById('uptime').textContent = formatUptime(metrics.uptime);
      document.getElementById('lastUpdate').textContent = formatDate(metrics.timestamp);

      // Update scheduled tasks
      document.getElementById('nextMigration').textContent = formatDate(metrics.scheduler.next_tier_migration);
      document.getElementById('nextBilling').textContent = formatDate(metrics.scheduler.next_billing_run);

      // Update billing pipeline
      document.getElementById('billingPending').textContent = metrics.billing.pending_count;
      document.getElementById('billingPendingAmount').textContent = `$${metrics.billing.pending_amount_usd}`;
      document.getElementById('billingCharged').textContent = '--';
      document.getElementById('billingChargedAmount').textContent = `$${metrics.billing.charged_this_month_usd}`;
      document.getElementById('billingFailed').textContent = metrics.billing.failed_count;

      // Update charts
      updateCharts(metrics);
    }

    function updateCharts(metrics) {
      // Tier distribution chart
      const tierCtx = document.getElementById('tierChart').getContext('2d');
      
      if (tierChart) {
        tierChart.data.datasets[0].data = Object.values(metrics.users.by_tier);
        tierChart.update();
      } else {
        tierChart = new Chart(tierCtx, {
          type: 'doughnut',
          data: {
            labels: ['Free', 'Individual', 'Pro', 'Team', 'Enterprise'],
            datasets: [{
              data: Object.values(metrics.users.by_tier),
              backgroundColor: [
                '#ef4444',
                '#f59e0b',
                '#10b981',
                '#3b82f6',
                '#8b5cf6'
              ],
            }],
          },
          options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
              legend: {
                labels: { color: '#e0e0e0' }
              }
            }
          }
        });
      }

      // Billing status chart
      const billingCtx = document.getElementById('billingChart').getContext('2d');
      
      if (billingChart) {
        billingChart.data.datasets[0].data = [
          metrics.billing.pending_count,
          metrics.billing.charged_this_month_usd / 100,
          metrics.billing.failed_count
        ];
        billingChart.update();
      } else {
        billingChart = new Chart(billingCtx, {
          type: 'bar',
          data: {
            labels: ['Pending', 'Charged (USD)', 'Failed'],
            datasets: [{
              label: 'Count / Amount',
              data: [
                metrics.billing.pending_count,
                metrics.billing.charged_this_month_usd / 100,
                metrics.billing.failed_count
              ],
              backgroundColor: ['#f59e0b', '#10b981', '#ef4444'],
            }],
          },
          options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
              legend: {
                labels: { color: '#e0e0e0' }
              }
            },
            scales: {
              y: {
                ticks: { color: '#8b8b9a' },
                grid: { color: 'rgba(255, 255, 255, 0.05)' }
              },
              x: {
                ticks: { color: '#8b8b9a' },
                grid: { color: 'rgba(255, 255, 255, 0.05)' }
              }
            }
          }
        });
      }
    }

    function manualRefresh() {
      const btn = event.target;
      btn.classList.add('loading');
      socket.emit('action:refresh');
      setTimeout(() => btn.classList.remove('loading'), 1000);
    }

    // Auto-scroll to latest update
    setInterval(() => {
      window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
    }, 30000);
  </script>
