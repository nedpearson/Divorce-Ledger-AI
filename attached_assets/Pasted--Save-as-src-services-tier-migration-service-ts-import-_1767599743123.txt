import { databaseService } from './database.service';
import { logger } from '../utils/logger';

export interface TierMigration {
  id: string;
  user_id: number;
  from_tier: string;
  to_tier: string;
  reason: string;
  grace_period_days: number; // Days before limits are enforced
  migrated_at: Date;
  effective_at: Date;
  status: 'pending' | 'active' | 'completed';
}

export class TierMigrationService {
  /**
   * Migrate user to new tier with optional grace period
   */
  async migrateTier(
    userId: number,
    newTier: string,
    reason: string = 'User requested',
    gracePeriodDays: number = 0
  ): Promise<TierMigration> {
    try {
      // Get current tier
      const userResult = await databaseService.query(
        `SELECT id, tier FROM users WHERE id = $1`,
        [userId]
      );

      if (!userResult.rows[0]) {
        throw new Error(`User ${userId} not found`);
      }

      const fromTier = userResult.rows[0].tier;

      if (fromTier === newTier) {
        throw new Error(`User already on ${newTier} tier`);
      }

      const now = new Date();
      const effectiveAt = new Date();
      effectiveAt.setDate(effectiveAt.getDate() + gracePeriodDays);

      const migration: TierMigration = {
        id: `migration_${userId}_${Date.now()}`,
        user_id: userId,
        from_tier: fromTier,
        to_tier: newTier,
        reason,
        grace_period_days: gracePeriodDays,
        migrated_at: now,
        effective_at: effectiveAt,
        status: gracePeriodDays > 0 ? 'pending' : 'active',
      };

      // Save migration record
      await databaseService.query(
        `INSERT INTO tier_migrations (
          id, user_id, from_tier, to_tier, reason, grace_period_days,
          migrated_at, effective_at, status
        ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)`,
        [
          migration.id,
          userId,
          fromTier,
          newTier,
          reason,
          gracePeriodDays,
          now,
          effectiveAt,
          migration.status,
        ]
      );

      // Update user tier if no grace period
      if (gracePeriodDays === 0) {
        await databaseService.query(`UPDATE users SET tier = $1 WHERE id = $2`, [newTier, userId]);
        logger.info(`✅ User ${userId} migrated: ${fromTier} → ${newTier}`);
      } else {
        logger.info(
          `⏳ User ${userId} migration scheduled: ${fromTier} → ${newTier} (grace period: ${gracePeriodDays} days)`
        );
      }

      return migration;
    } catch (error) {
      logger.error('Tier migration failed', error);
      throw error;
    }
  }

  /**
   * Apply pending migrations that have passed grace period
   */
  async applyPendingMigrations(): Promise<{ applied: number; failed: number }> {
    try {
      const result = await databaseService.query(
        `SELECT id, user_id, to_tier FROM tier_migrations
         WHERE status = 'pending' AND effective_at <= NOW()`
      );

      let applied = 0;
      let failed = 0;

      for (const migration of result.rows) {
        try {
          await databaseService.query(`UPDATE users SET tier = $1 WHERE id = $2`, [
            migration.to_tier,
            migration.user_id,
          ]);

          await databaseService.query(
            `UPDATE tier_migrations SET status = 'active' WHERE id = $1`,
            [migration.id]
          );

          applied++;
          logger.info(`✅ Applied pending migration for user ${migration.user_id}`);
        } catch (error) {
          logger.error(`Failed to apply migration for user ${migration.user_id}`, error);
          failed++;
        }
      }

      return { applied, failed };
    } catch (error) {
      logger.error('Failed to apply pending migrations', error);
      throw error;
    }
  }

  /**
   * Get active migration for user (if any)
   */
  async getActiveMigration(userId: number): Promise<TierMigration | null> {
    try {
      const result = await databaseService.query(
        `SELECT * FROM tier_migrations
         WHERE user_id = $1 AND status IN ('pending', 'active')
         ORDER BY migrated_at DESC LIMIT 1`,
        [userId]
      );

      return result.rows[0] || null;
    } catch (error) {
      logger.error('Failed to get active migration', error);
      throw error;
    }
  }

  /**
   * Cancel pending tier migration
   */
  async cancelMigration(migrationId: string): Promise<void> {
    try {
      await databaseService.query(
        `UPDATE tier_migrations SET status = 'completed' WHERE id = $1`,
        [migrationId]
      );

      logger.info(`✅ Migration ${migrationId} cancelled`);
    } catch (error) {
      logger.error('Failed to cancel migration', error);
      throw error;
    }
  }
}

export const tierMigrationService = new TierMigrationService();
