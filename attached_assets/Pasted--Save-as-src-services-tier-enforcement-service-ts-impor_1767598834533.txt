import { databaseService } from './database.service';
import { logger } from '../utils/logger';

export interface TierLimits {
  tier: string;
  maxFileSizeMB: number;
  maxStorageMB: number | null; // null = unlimited
  maxViolationsPerMonth: number | null; // null = unlimited
  maxCases: number | null; // null = unlimited
}

export interface UserUsageMetrics {
  user_id: number;
  tier: string;
  violations_this_month: number;
  storage_used_mb: number;
  media_count: number;
  file_uploads_this_month: number;
  cases_active: number;
}

export interface TierEnforcementResult {
  allowed: boolean;
  reason?: string;
  current_usage: UserUsageMetrics;
  tier_limits: TierLimits;
  warning?: string;
}

const TIER_LIMITS: Record<string, TierLimits> = {
  free: {
    tier: 'free',
    maxFileSizeMB: 10,
    maxStorageMB: 100,
    maxViolationsPerMonth: 10,
    maxCases: 1,
  },
  individual: {
    tier: 'individual',
    maxFileSizeMB: 50,
    maxStorageMB: 500,
    maxViolationsPerMonth: 20,
    maxCases: 1,
  },
  pro: {
    tier: 'pro',
    maxFileSizeMB: 100,
    maxStorageMB: 2048, // 2GB
    maxViolationsPerMonth: 50,
    maxCases: null, // Unlimited
  },
  team: {
    tier: 'team',
    maxFileSizeMB: 250,
    maxStorageMB: 10240, // 10GB
    maxViolationsPerMonth: null, // Unlimited
    maxCases: null, // Unlimited
  },
  enterprise: {
    tier: 'enterprise',
    maxFileSizeMB: 500,
    maxStorageMB: null, // Unlimited
    maxViolationsPerMonth: null, // Unlimited
    maxCases: null, // Unlimited
  },
};

export class TierEnforcementService {
  /**
   * Check if user can upload a file based on tier limits
   */
  async canUploadFile(userId: number, fileSizeMB: number): Promise<TierEnforcementResult> {
    try {
      const userResult = await databaseService.query(
        `SELECT tier FROM users WHERE id = $1`,
        [userId]
      );

      if (userResult.rows.length === 0) {
        throw new Error(`User ${userId} not found`);
      }

      const userTier = userResult.rows[0].tier;
      const tierLimits = TIER_LIMITS[userTier];
      const currentUsage = await this.getUserUsageMetrics(userId);

      // Check 1: File size limit
      if (fileSizeMB > tierLimits.maxFileSizeMB) {
        return {
          allowed: false,
          reason: `File size (${fileSizeMB}MB) exceeds tier limit (${tierLimits.maxFileSizeMB}MB)`,
          current_usage: currentUsage,
          tier_limits: tierLimits,
        };
      }

      // Check 2: Monthly storage limit
      if (tierLimits.maxStorageMB !== null) {
        const storageRemaining = tierLimits.maxStorageMB - currentUsage.storage_used_mb;

        if (fileSizeMB > storageRemaining) {
          return {
            allowed: false,
            reason: `File would exceed monthly storage limit. Used: ${currentUsage.storage_used_mb}MB / ${tierLimits.maxStorageMB}MB`,
            current_usage: currentUsage,
            tier_limits: tierLimits,
          };
        }
      }

      // Check 3: Monthly violation limit
      if (tierLimits.maxViolationsPerMonth !== null) {
        if (currentUsage.violations_this_month >= tierLimits.maxViolationsPerMonth) {
          return {
            allowed: false,
            reason: `Monthly violation limit reached (${currentUsage.violations_this_month}/${tierLimits.maxViolationsPerMonth})`,
            current_usage: currentUsage,
            tier_limits: tierLimits,
          };
        }
      }

      // All checks passed - but warn if approaching limits
      const warning = this.generateUsageWarning(currentUsage, tierLimits);

      return {
        allowed: true,
        current_usage: currentUsage,
        tier_limits: tierLimits,
        warning,
      };
    } catch (error) {
      logger.error('Tier enforcement check failed', error);
      throw error;
    }
  }

  /**
   * Get comprehensive usage metrics for a user
   */
  async getUserUsageMetrics(userId: number): Promise<UserUsageMetrics> {
    try {
      // Get violations count
      const violationsResult = await databaseService.query(
        `SELECT COUNT(*) as count
         FROM violations v
         JOIN cases c ON v.case_id = c.id
         WHERE c.lawyer_id = $1
         AND DATE_TRUNC('month', v.created_at) = DATE_TRUNC('month', NOW())`,
        [userId]
      );

      // Get storage and media count
      const storageResult = await databaseService.query(
        `SELECT 
          SUM(e.file_size_bytes) / (1024*1024) as storage_mb,
          COUNT(DISTINCT e.id) as media_count,
          COUNT(DISTINCT DATE(e.created_at)) as days_with_uploads
         FROM evidence e
         JOIN violations v ON e.violation_id = v.id
         JOIN cases c ON v.case_id = c.id
         WHERE c.lawyer_id = $1
         AND e.evidence_source = 'media_upload'
         AND DATE_TRUNC('month', e.created_at) = DATE_TRUNC('month', NOW())`,
        [userId]
      );

      // Get active cases count
      const casesResult = await databaseService.query(
        `SELECT COUNT(DISTINCT id) as case_count
         FROM cases
         WHERE lawyer_id = $1
         AND status != 'closed'`,
        [userId]
      );

      // Get user tier
      const userResult = await databaseService.query(
        `SELECT tier FROM users WHERE id = $1`,
        [userId]
      );

      const storage = storageResult.rows[0];
      const user = userResult.rows[0];

      return {
        user_id: userId,
        tier: user.tier,
        violations_this_month: parseInt(violationsResult.rows[0].count) || 0,
        storage_used_mb: parseFloat(storage.storage_mb) || 0,
        media_count: parseInt(storage.media_count) || 0,
        file_uploads_this_month: storage.days_with_uploads ? parseInt(storage.days_with_uploads) : 0,
        cases_active: parseInt(casesResult.rows[0].case_count) || 0,
      };
    } catch (error) {
      logger.error('Failed to get usage metrics', error);
      throw error;
    }
  }

  /**
   * Generate usage warnings based on tier limits
   */
  private generateUsageWarning(usage: UserUsageMetrics, limits: TierLimits): string | undefined {
    const warnings: string[] = [];

    // Warning: Approaching file size
    const fileSizePercentage = 0; // Can't easily check without context of all files

    // Warning: Storage threshold
    if (limits.maxStorageMB !== null) {
      const storagePercentage = (usage.storage_used_mb / limits.maxStorageMB) * 100;
      if (storagePercentage > 80) {
        warnings.push(
          `⚠️ Storage usage at ${storagePercentage.toFixed(0)}% (${usage.storage_used_mb.toFixed(1)}MB / ${limits.maxStorageMB}MB)`
        );
      }
    }

    // Warning: Violation threshold
    if (limits.maxViolationsPerMonth !== null) {
      const violationPercentage = (usage.violations_this_month / limits.maxViolationsPerMonth) * 100;
      if (violationPercentage > 80) {
        warnings.push(
          `⚠️ Violation limit at ${violationPercentage.toFixed(0)}% (${usage.violations_this_month}/${limits.maxViolationsPerMonth})`
        );
      }
    }

    // Warning: Cases threshold
    if (limits.maxCases !== null) {
      const casePercentage = (usage.cases_active / limits.maxCases) * 100;
      if (casePercentage > 80) {
        warnings.push(
          `⚠️ Case limit at ${casePercentage.toFixed(0)}% (${usage.cases_active}/${limits.maxCases})`
        );
      }
    }

    return warnings.length > 0 ? warnings.join(' | ') : undefined;
  }

  /**
   * Get recommended tier upgrade for user
   */
  async getRecommendedTierUpgrade(userId: number): Promise<{
    current_tier: string;
    recommended_tier: string;
    reason: string;
    upgrade_cost_savings?: string;
  }> {
    try {
      const usage = await this.getUserUsageMetrics(userId);
      const currentLimits = TIER_LIMITS[usage.tier];

      // Check what tier they should be on
      let recommendedTier = usage.tier;
      let reason = 'Within current tier limits';

      // Check if current tier is insufficient
      if (
        currentLimits.maxViolationsPerMonth !== null &&
        usage.violations_this_month >= currentLimits.maxViolationsPerMonth * 0.8
      ) {
        recommendedTier = this.getNextTier(usage.tier);
        reason = `Approaching violation limit (${usage.violations_this_month}/${currentLimits.maxViolationsPerMonth})`;
      }

      if (
        currentLimits.maxStorageMB !== null &&
        usage.storage_used_mb >= currentLimits.maxStorageMB * 0.8
      ) {
        const tierForStorage = this.getTierForStorage(usage.storage_used_mb);
        if (this.tierHierarchy(tierForStorage) > this.tierHierarchy(recommendedTier)) {
          recommendedTier = tierForStorage;
          reason = `Approaching storage limit (${usage.storage_used_mb.toFixed(1)}MB/${currentLimits.maxStorageMB}MB)`;
        }
      }

      if (
        currentLimits.maxCases !== null &&
        usage.cases_active >= currentLimits.maxCases * 0.8
      ) {
        const tierForCases = this.getTierForCases(usage.cases_active);
        if (this.tierHierarchy(tierForCases) > this.tierHierarchy(recommendedTier)) {
          recommendedTier = tierForCases;
          reason = `Approaching case limit (${usage.cases_active}/${currentLimits.maxCases})`;
        }
      }

      return {
        current_tier: usage.tier,
        recommended_tier: recommendedTier,
        reason,
        upgrade_cost_savings:
          usage.tier !== recommendedTier
            ? `Upgrade to ${recommendedTier} for unlimited resources`
            : undefined,
      };
    } catch (error) {
      logger.error('Failed to get recommended tier', error);
      throw error;
    }
  }

  /**
   * Log usage metrics to usage_audit table for analytics
   */
  async logUsageMetrics(userId: number): Promise<void> {
    try {
      const usage = await this.getUserUsageMetrics(userId);

      await databaseService.query(
        `INSERT INTO usage_audit (
          user_id, tier, violations_count, storage_used_mb, media_count, 
          active_cases, recorded_at
        ) VALUES ($1, $2, $3, $4, $5, $6, NOW())`,
        [
          userId,
          usage.tier,
          usage.violations_this_month,
          usage.storage_used_mb,
          usage.media_count,
          usage.cases_active,
        ]
      );

      logger.info(`📊 Usage logged for user ${userId}`);
    } catch (error) {
      logger.error('Failed to log usage metrics', error);
      // Don't throw - this is non-critical
    }
  }

  /**
   * Get tier hierarchy for comparison
   */
  private tierHierarchy(tier: string): number {
    const hierarchy: Record<string, number> = {
      free: 0,
      individual: 1,
      pro: 2,
      team: 3,
      enterprise: 4,
    };
    return hierarchy[tier] || 0;
  }

  /**
   * Get next tier up
   */
  private getNextTier(currentTier: string): string {
    const tiers = ['free', 'individual', 'pro', 'team', 'enterprise'];
    const index = tiers.indexOf(currentTier);
    return index < tiers.length - 1 ? tiers[index + 1] : 'enterprise';
  }

  /**
   * Get minimum tier needed for storage
   */
  private getTierForStorage(storageMB: number): string {
    if (storageMB > 10240) return 'enterprise'; // > 10GB
    if (storageMB > 2048) return 'team'; // > 2GB
    if (storageMB > 500) return 'pro'; // > 500MB
    if (storageMB > 100) return 'individual'; // > 100MB
    return 'free';
  }

  /**
   * Get minimum tier needed for cases
   */
  private getTierForCases(caseCount: number): string {
    if (caseCount > 1) return 'pro'; // Unlimited from pro+
    return 'free';
  }

  /**
   * Get tier limits (public helper)
   */
  static getTierLimits(tier: string): TierLimits {
    return TIER_LIMITS[tier] || TIER_LIMITS.free;
  }

  /**
   * Check if tier can have multiple cases
   */
  static canHaveMultipleCases(tier: string): boolean {
    return tier !== 'free' && tier !== 'individual';
  }
}

export const tierEnforcementService = new TierEnforcementService();
