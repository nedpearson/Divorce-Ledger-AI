
2. Create audit log table:
```sql
CREATE TABLE IF NOT EXISTS quickbooks_sync_log (
  id SERIAL PRIMARY KEY,
  user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  action VARCHAR(100) NOT NULL,
  qb_entity_type VARCHAR(50),
  qb_entity_id VARCHAR(50),
  request_method VARCHAR(10),
  request_path TEXT,
  response_status INTEGER,
  error_message TEXT,
  metadata JSONB,
  created_at TIMESTAMP DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_qb_log_user ON quickbooks_sync_log(user_id);
CREATE INDEX IF NOT EXISTS idx_qb_log_created ON quickbooks_sync_log(created_at);
```

3. Link cases to QB customers:
```sql
ALTER TABLE cases ADD COLUMN IF NOT EXISTS qb_customer_id VARCHAR(50);
ALTER TABLE cases ADD COLUMN IF NOT EXISTS qb_customer_name VARCHAR(255);
```
